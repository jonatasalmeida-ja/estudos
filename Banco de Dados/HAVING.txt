# Revisão do Having
SELECT
	genre,
    COUNT(*) total_songs,
    ROUND(AVG(duration_seconds), 1) avg_duration
FROM songs
WHERE year > 2000
GROUP BY genre
HAVING COUNT(*) > 3
ORDER BY total_songs DESC;

SELECT
	genre,
    COUNT(*) total_songs,
    ROUND(AVG(duration_seconds), 1) avg_duration
FROM songs
WHERE year > 2000
GROUP BY genre
HAVING COUNT(*) > 2
ORDER BY total_songs DESC;

SELECT
	genre,
	COUNT(*) total_songs,
    ROUND(AVG(duration_seconds), 1) avg_duration
FROM songs
WHERE duration_seconds > 120
GROUP BY genre
ORDER BY avg_duration DESC;

SELECT
	genre,
    COUNT(*) total_songs,
    MAX(duration_seconds) max_duration
FROM songs
WHERE year > 2010
GROUP BY genre
HAVING COUNT(*) > 3
ORDER BY max_duration DESC;
